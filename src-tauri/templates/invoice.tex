\newcommand{\naturalwidth}[2]{\settowidth{#1}{\renewcommand{\hfill}{\enskip}\begin{tabular}{@{}l@{}}#2\end{tabular}}}
\newcommand{\widestword}[2]{%
  \setbox0\vbox{\hsize=0pt\addressblock\renewcommand{\\}[1][]{ }#2\par
    \global#1=0pt
    \loop\unskip\unpenalty\setbox2\lastbox\ifhbox2
      \setbox2\hbox{\unhbox2}%
      \ifdim\wd2>#1\global#1=\wd2\fi
    \repeat}}
\newcommand{\fitaddresscolumns}{%
  \naturalwidth{\fromwidth}{\addressfrom}%
  \naturalwidth{\towidth}{\addressto}%
  \naturalwidth{\detailswidth}{\addressdetails}%
  \widestword{\fromword}{\addressfrom}%
  \widestword{\toword}{\addressto}%
  \setlength{\addressspace}{\dimexpr\linewidth-4\tabcolsep\relax}%
  \pgfmathsetlength{\detailswidth}{min(\detailswidth,max(\addressspace/3,\addressspace-\fromword-\toword))}%
  \setlength{\addressrest}{\dimexpr\addressspace-\detailswidth\relax}%
  \pgfmathsetlength{\fromwidth}{min(max(\addressrest/2,\addressrest-\toword),max(\fromword,min(\fromwidth,max(\addressrest/2,\addressrest-\towidth))))}%
  \pgfmathsetlength{\towidth}{min(\towidth,\addressrest-\fromwidth)}%
  \pgfmathsetlength{\addressshare}{min(\addressspace/3,(\addressspace-\fromwidth)/2,(\addressspace-\towidth)/2,(\addressspace-\detailswidth)/2,\addressspace-\fromwidth-\towidth,\addressspace-\fromwidth-\detailswidth,\addressspace-\towidth-\detailswidth)}%
  \pgfmathsetlength{\fromwidth}{max(\fromwidth,\addressshare)}%
  \pgfmathsetlength{\towidth}{max(\towidth,\addressshare)}%
  \setlength{\detailswidth}{\dimexpr\addressspace-\fromwidth-\towidth\relax}%
}
